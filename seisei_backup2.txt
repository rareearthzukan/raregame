    function getRandomElementKey() {
      const keys = Object.keys(elementsMaster);
      let pool = [];
      keys.forEach(k => {
        if (elementsMaster[k].type === 'light') {
          pool.push(k, k, k);
        } else {
          pool.push(k);
        }
      });
      return pool[Math.floor(Math.random() * pool.length)];
    }

    function refineElement() {
      if (gameState.mudStock <= 0) {
        showShortageModal('mud');
        return;
      }
      if (gameState.money < 50) {
        showShortageModal('money');
        return;
      }

      const soundId = Math.random() < 0.5 ? 'se-nyu1' : 'se-nyu2';
      const se = document.getElementById(soundId);
      if (se) {
        se.currentTime = 0;
        se.play().catch(e => console.log('音声再生エラー:', e));
      }

      gameState.money -= 50;
      gameState.mudStock--;

      const targetKey = getRandomElementKey();

      // 約10%の確率で10倍ハイ＆ローイベント発動
      const roll = Math.random();
      if (roll < 0.10) {
        showHighLowMinigame(targetKey);
      } else {
        executeNormalRefine(targetKey, 1);
      }
    }

    function evalHighLow(targetKey, baseNum, baseSymbol, secretNum, secretName, secretSymbol, playerGuess) {
      let isActualHigh = secretNum > baseNum;
      let playerGuessHigh = (playerGuess === 'high');
      let isWin = (playerGuessHigh === isActualHigh);

      let mult = isWin ? 10 : 0;
      executeNormalRefine(targetKey, mult);

      // サウンド再生処理
      if (isWin) {
        const itemGetSe = document.getElementById('se-itemget');
        if (itemGetSe) {
          itemGetSe.currentTime = 0;
          itemGetSe.play().catch(e => {});
        }
      } else {
        const fuseikaiSe = document.getElementById('se-fuseikai');
        if (fuseikaiSe) {
          fuseikaiSe.currentTime = 0;
          fuseikaiSe.play().catch(e => {});
        }
      }

      // 原子番号の大小に応じて大きい方の大ボックス、小さい方の小ボックスに割り当て
      let leftSymbol, leftNum, rightSymbol, rightNum;
      if (secretNum >= baseNum) {
        leftSymbol = secretSymbol; leftNum = secretNum;
        rightSymbol = elementsMaster[baseSymbol].symbol; rightNum = baseNum;
      } else {
        leftSymbol = elementsMaster[baseSymbol].symbol; leftNum = baseNum;
        rightSymbol = secretSymbol; rightNum = secretNum;
      }

      const content = document.getElementById('modal-content-inner');
      content.innerHTML = `
        <div style="font-size: 24px;">${isWin ? '🌟🌟🌟' : '💥'}</div>
        <div style="font-size: 12px; color: #00fff5; margin: 2px 0; font-weight: bold;">
          ${secretName}が現れました！
        </div>
        <div style="font-size: 13px; font-weight: bold; color: ${isWin ? '#ffde59' : '#ff4757'}; margin: 2px 0; line-height: 1.3;">
          ${isWin ? '大正解！10倍ＧＥＴです！' : '残念…負けてしまいました（汗）<br>今回は０個ＧＥＴです。'}
        </div>
        
        <!-- 勝敗ビジュアル化ボックス（原子番号の大きい方を大ボックス、小さい方を小ボックスに配置） -->
        <div class="visual-comparison">
          <div class="box-large" title="大きい原子番号">
            <span style="font-size:15px;">${leftSymbol}</span>
            <span style="font-size:11px; color:#fff;">${leftNum}</span>
          </div>
          <div style="font-size:14px; color:#8a8a8a; font-weight:bold;">VS</div>
          <div class="box-small" title="小さい原子番号">
            <span style="font-size:12px;">${rightSymbol}</span>
            <span style="font-size:9px; color:#ddd;">${rightNum}</span>
          </div>
        </div>

        <button class="btn" style="margin-top:4px;" onclick="closeMinigameModal()">閉じる</button>
      `;
    }
